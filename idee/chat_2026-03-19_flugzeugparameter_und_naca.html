import numpy as np
import matplotlib.pyplot as plt

# =======================
# Konstanten / Parameter
# =======================
CHORD_LENGTH_M = 0.2 # Flügeltiefe der Rippe in Metern
NUM_POINTS = 50 # Anzahl Punkte zur Profilkontur
NACA_MAX_CAMBER = 0.02 # maximale Wölbung (m)
NACA_CAMBER_POS = 0.4 # Position der maximalen Wölbung (p)
NACA_THICKNESS = 0.12 # maximale Dicke (t)
OUTPUT_PLOT = True # True: zeigt Plot der Rippe
OUTPUT_SVG = "rippe.svg" # Dateiname für SVG
OUTPUT_DXF = "rippe.dxf" # Dateiname für DXF

# =======================
# Funktionen
# =======================
def generate_naca4_airfoil(max_camber, camber_pos, thickness, chord=1.0, points=50):
    """
    Berechnet x,y-Koordinaten für ein NACA4-Profil
    """
    x = np.linspace(0, chord, points)
    # Dickenverteilung
    yt = 5 * thickness * chord * (
        0.2969 * np.sqrt(x/chord)
        - 0.1260 * (x/chord)
        - 0.3516 * (x/chord)**2
        + 0.2843 * (x/chord)**3
        - 0.1015 * (x/chord)**4
    )
    # Mittellinie
    yc = np.zeros_like(x)
    for i in range(points):
        if x[i] < camber_pos * chord:
            yc[i] = max_camber / camber_pos**2 * (2*camber_pos*(x[i]/chord) - (x[i]/chord)**2)
        else:
            yc[i] = max_camber / (1-camber_pos)**2 * ((1-2*camber_pos) + 2*camber_pos*(x[i]/chord) - (x[i]/chord)**2)

    theta = np.arctan(np.gradient(yc, x))
    # obere und untere Fläche
    xu = x - yt * np.sin(theta)
    yu = yc + yt * np.cos(theta)
    xl = x + yt * np.sin(theta)
    yl = yc - yt * np.cos(theta)

    # Verbinden für durchgehende Kontur (Tip->Root->Tip)
    x_full = np.concatenate([xu[::-1], xl[1:]])
    y_full = np.concatenate([yu[::-1], yl[1:]])
    return x_full, y_full

def plot_airfoil(x_coords, y_coords):
    plt.plot(x_coords, y_coords, color="blue")
    plt.axis("equal")
    plt.title("NACA4 Rippe")
    plt.xlabel("x [m]")
    plt.ylabel("y [m]")
    plt.show()

# =======================
# Hauptprogramm
# =======================
if __name__ == "__main__":
    x_coords, y_coords = generate_naca4_airfoil(
        max_camber=NACA_MAX_CAMBER,
        camber_pos=NACA_CAMBER_POS,
        thickness=NACA_THICKNESS,
        chord=CHORD_LENGTH_M,
        points=NUM_POINTS
    )

    if OUTPUT_PLOT:
        plot_airfoil(x_coords, y_coords)

    # Optional: SVG export
    try:
        import svgwrite
        dwg = svgwrite.Drawing(OUTPUT_SVG, profile='tiny', size=(f"{CHORD_LENGTH_M*1000}mm", f"{CHORD_LENGTH_M*1000}mm"))
        points_svg = [(x*1000, -y*1000) for x, y in zip(x_coords, y_coords)] # mm, y invertiert
        dwg.add(dwg.polyline(points_svg, stroke='black', fill='none', stroke_width=1))
        dwg.save()
        print(f"SVG gespeichert: {OUTPUT_SVG}")
    except ImportError:
        print("svgwrite nicht installiert, SVG wird nicht erzeugt.")

    # Optional: DXF export
    try:
        import ezdxf
        doc = ezdxf.new()
        msp = doc.modelspace()
        for i in range(len(x_coords)-1):
            msp.add_line((x_coords[i]*1000, y_coords[i]*1000), (x_coords[i+1]*1000, y_coords[i+1]*1000))
        doc.saveas(OUTPUT_DXF)
        print(f"DXF gespeichert: {OUTPUT_DXF}")
    except ImportError:
        print("ezdxf nicht installiert, DXF wird nicht erzeugt.")

def estimate_chord(area, span):

    Schätzt die mittlere Flügeltiefe (Chord) aus Fläche und Spannweite
    A = b * c_mean
    """
    return area / span

def estimate_naca_parameters(design_cl=0.5, thickness_ratio=0.12):

✈️ NACA 4-stellige Profile lesen (z. B. NACA 2412)

Format: 👉 NACA m p tt

Stelle Bedeutung
m maximale Wölbung in % der Flügeltiefe
p Position dieser Wölbung in Zehnteln der Tiefe
tt maximale Dicke in %

🔍 Beispiel: NACA 2412

  • 2 → 2 % Wölbung
  • 4 → bei 40 % der Tiefe
  • 12 → 12 % Dicke

👉 Übersetzt:

leicht gewölbt, Maximum relativ weit vorne, moderat dick

🧠 So kannst du es dir merken

  • erste Zahl (m) → „wie stark gebogen?“
  • zweite Zahl (p) → „wo ist der Bauch?“
  • letzte zwei (tt) → „wie dick ist das Profil?“

⚖️ Typische Auswirkungen

1. Wölbung (m)

  • klein (0–2 %) → schnell, wenig Auftrieb
  • mittel (3–5 %) → guter Allrounder
  • groß (>6 %) → viel Auftrieb, viel Widerstand

2. Position (p)

  • vorne (z. B. 2–3) → stabil, gutmütig
  • mittig (4–5) → Standard
  • hinten → selten, eher speziell

3. Dicke (tt)

  • 6–9 % → schnell, wenig Widerstand
  • 10–15 % → stabil, gut für Modelle

  • 15 % → langsam, viel Auftrieb

🔥 Noch ein paar Beispiele

  • NACA 0012 0 % Wölbung → symmetrisch (Kunstflug)

  • NACA 4412 4 % Wölbung → viel Auftrieb (Trainer)

  • NACA 2412 Klassiker → Allround-Profil

✈️ 1. Das theoretische Optimum

🟢 Elliptische Auftriebsverteilung

👉 Minimierter induzierter Widerstand (wichtig bei kleinen Flugzeugen!)

  • sehr effizient

❗ Aber:

  • schwer zu bauen
  • empfindlich
  • strukturell schlechter
  • kaum besser als gute Näherungen

⚖️ 2. Praxis: Was ist wirklich optimal?

👉 Statt „perfekt elliptisch“ nimmt man:

🔵 Trapezflügel (Standard!)

  • einfach zu bauen
  • fast genauso effizient

📐 3. Entscheidender Parameter: Zuspitzung (Taper Ratio)

[ \lambda = \frac{c_{tip}}{c_{root}} ]

💡 Optimale Werte:

Ziel λ (Tip/Root)
minimaler Widerstand ~0.3–0.5
stabil & gutmütig ~0.5–0.7
einfach bauen ~1.0 (rechteckig)

🚀 4. Empfehlung für dein Projekt

👉 Beste Wahl für JuFo:

  • λ ≈ 0.4–0.6
  • also:
  • Spitze ≈ 40–60 % der Wurzel

📊 5. Beispielrechnung

Gegeben: - Spannweite: 1.2 m - Fläche: 0.18 m²

Mittlere Tiefe: [ c_{mean} = A / b = 0.15\,m ]

Mit λ = 0.5:

👉 Formeln: [ c_{root} = \frac{2A}{b(1+\lambda)} ] [ c_{tip} = \lambda \cdot c_{root} ]

Ergebnis: - Root ≈ 0.20 m - Tip ≈ 0.10 m

👉 Sehr guter Wert!

🔧 6. Code dafür

def wing_geometry(area, span, taper_ratio=0.5):
    """
    Berechnet Root- und Tip-Chord
    """
    c_root = 2 * area / (span * (1 + taper_ratio))
    c_tip = taper_ratio * c_root
    return c_root, c_tip
